name: full install/update

on: [push]

jobs:
  test-script:
    runs-on: ubuntu-latest
    env:
      CONTAINER_NAME: spartan-test-${{ github.run_id }}
    steps:
      - name: Check out repository
        uses: actions/checkout@v6

      - name: Start Ubuntu Systemd Container
        run: |
          mkdir -p ./ci-cache/apt ./ci-cache/npm ./ci-cache/composer
          docker run -d --name $CONTAINER_NAME \
            -p 80:80 -p 443:443 \
            --privileged \
            --cgroupns=host \
            --tmpfs /tmp \
            --tmpfs /run \
            --tmpfs /run/lock \
            -v /sys/fs/cgroup:/sys/fs/cgroup:rw \
            -v /tmp/spartan-ci-cache/apt:/var/cache/apt/archives \
            -v /tmp/spartan-ci-cache/npm:/root/.npm \
            -v /tmp/spartan-ci-cache/composer:/root/.composer/cache \
            jrei/systemd-ubuntu:22.04

          echo "Waiting for systemd to be ready..."
          TIMEOUT=60
          ELAPSED=0
          while [ $ELAPSED -lt $TIMEOUT ]; do
            STATUS=$(docker exec $CONTAINER_NAME systemctl is-system-running 2>/dev/null || echo "not-ready")
            if [ "$STATUS" = "running" ] || [ "$STATUS" = "degraded" ]; then
              echo "Systemd is ready (status: $STATUS) in $ELAPSED seconds!"
              break
            fi

            echo "Status is '$STATUS'. Waiting..."
            sleep 2
            ELAPSED=$((ELAPSED+2))
          done

          if [ $ELAPSED -ge $TIMEOUT ]; then
            echo "Error: systemd did not start within $TIMEOUT seconds."
            docker logs $CONTAINER_NAME
            exit 1
          fi

      - name: Copy files to container
        run: docker cp . $CONTAINER_NAME:/opt/spartan-installer

      - name: Test install
        env:
          SPARTAN_LICENSE: ${{ secrets.SPARTAN_LICENSE }}
        run: |
          docker exec $CONTAINER_NAME bash -c "cd /opt/spartan-installer && chmod +x ./spartan_installer.sh && bash spartan_installer.sh --non-interactive --install --license=$SPARTAN_LICENSE --domain spartan.test.blipzoo.ovh --webserver=nginx --ssl-mode=assume --db-type=mariadb"

      - name: Smoke Test (Check if site is alive)
        run: |
          echo "Waiting for the dashboard to come online..."
          TIMEOUT=60
          ELAPSED=0
          while [ $ELAPSED -lt $TIMEOUT ]; do
            STATUS=$(docker exec $CONTAINER_NAME curl -s -o /dev/null -w "%{http_code}" -H "Host: spartan.test.blipzoo.ovh" --insecure https://127.0.0.1 || echo "000")
            
            if [[ "$STATUS" =~ ^(200|301|302)$ ]]; then
              echo "✅ Dashboard is alive! (HTTP $STATUS) in $ELAPSED seconds."
              exit 0
            fi
            
            echo "Dashboard returned HTTP $STATUS. Waiting..."
            sleep 5
            ELAPSED=$((ELAPSED+5))
          done
          
          echo "❌ Dashboard failed to come online within $TIMEOUT seconds."
          docker exec $CONTAINER_NAME tail -n 50 /var/www/spartan/storage/logs/laravel.log
          exit 1

      - name: Take Dashboard Screenshot
        run: |
          docker exec $CONTAINER_NAME bash -c "cd /opt/spartan-installer/.gitea/workflows/tools && chmod +x ./take-screenshot.sh && bash ./take-screenshot.sh"
          echo "Extracting screenshot from container..."
          docker cp $CONTAINER_NAME:/tmp/dashboard.png ./dashboard.png || echo "Failed to extract screenshot."

      - name: Upload Screenshot Artifact
        if: always()
        uses: https://github.com/christopherHX/gitea-upload-artifact@v4
        with:
          name: spartan-before-login-screenshot
          path: ./dashboard.png
          if-no-files-found: ignore

      - name: Extract Log File (before update)
        if: always()
        run: docker cp $CONTAINER_NAME:/var/log/spartan_installer.log ./spartan_before_installer.log || true

      - name: Extract Spartan Files (before update)
        run: |
          if docker exec $CONTAINER_NAME test -d /var/www/spartan; then
            echo "App directory found! Zipping..."
            docker exec $CONTAINER_NAME tar -czf /tmp/spartan_before_files.tar.gz -C /var/www spartan || true
            docker cp $CONTAINER_NAME:/tmp/spartan_before_files.tar.gz ./spartan_before_files.tar.gz || true
          else
            echo "App directory /var/www/spartan does not exist yet. Skipping zip."
          fi

      - name: Test Update
        run: |
          docker exec $CONTAINER_NAME bash -c "cd /opt/spartan-installer && bash spartan_installer.sh --non-interactive --update"

      - name: Smoke Test (Check if site is alive)
        run: |
          echo "Waiting for the dashboard to come online..."
          TIMEOUT=60
          ELAPSED=0
          while [ $ELAPSED -lt $TIMEOUT ]; do
            STATUS=$(docker exec $CONTAINER_NAME curl -s -o /dev/null -w "%{http_code}" -H "Host: spartan.test.blipzoo.ovh" --insecure https://127.0.0.1 || echo "000")
            
            if [[ "$STATUS" =~ ^(200|301|302)$ ]]; then
              echo "✅ Dashboard is alive! (HTTP $STATUS) in $ELAPSED seconds."
              exit 0
            fi
            
            echo "Dashboard returned HTTP $STATUS. Waiting..."
            sleep 5
            ELAPSED=$((ELAPSED+5))
          done
          
          echo "❌ Dashboard failed to come online within $TIMEOUT seconds."
          docker exec $CONTAINER_NAME tail -n 50 /var/www/spartan/storage/logs/laravel.log
          exit 1

      - name: Take Dashboard Screenshot
        run: |
          docker exec $CONTAINER_NAME bash -c "cd /opt/spartan-installer/.gitea/workflows && chmod +x ./take-screenshot.sh && bash ./take-screenshot.sh"
          echo "Extracting screenshot from container..."
          docker cp $CONTAINER_NAME:/tmp/dashboard.png ./dashboard.png || echo "Failed to extract screenshot."

      - name: Upload Screenshot Artifact
        if: always()
        uses: https://github.com/christopherHX/gitea-upload-artifact@v4
        with:
          name: spartan-after-login-screenshot
          path: ./dashboard.png
          if-no-files-found: ignore

      - name: Extract Log File (after update)
        if: always()
        run: docker cp $CONTAINER_NAME:/var/log/spartan_installer.log ./spartan_after_installer.log || true

      - name: Extract Spartan Files (after update)
        run: |
          if docker exec $CONTAINER_NAME test -d /var/www/spartan; then
            echo "App directory found! Zipping..."
            docker exec $CONTAINER_NAME tar -czf /tmp/spartan_after_files.tar.gz -C /var/www spartan || true
            docker cp $CONTAINER_NAME:/tmp/spartan_after_files.tar.gz ./spartan_after_files.tar.gz || true
          else
            echo "App directory /var/www/spartan does not exist yet. Skipping zip."
          fi

      - name: Upload Installer Log (before)
        if: always()
        uses: https://github.com/christopherHX/gitea-upload-artifact@v4
        with:
          name: spartan-before-installer-log
          path: ./spartan_before_installer.log
          if-no-files-found: warn
      
      - name: Upload Installer Log (after)
        if: always()
        uses: https://github.com/christopherHX/gitea-upload-artifact@v4
        with:
          name: spartan-after-installer-log
          path: ./spartan_after_installer.log
          if-no-files-found: warn

      - name: Upload Files Artifact (before)
        uses: https://github.com/christopherHX/gitea-upload-artifact@v4
        with:
          name: spartan-before-installed-files
          path: ./spartan_before_files.tar.gz
          if-no-files-found: ignore

      - name: Upload Files Artifact (after)
        uses: https://github.com/christopherHX/gitea-upload-artifact@v4
        with:
          name: spartan-after-installed-files
          path: ./spartan_after_files.tar.gz
          if-no-files-found: ignore

      - name: Cleanup Test Environment
        if: always()
        run: docker rm -f $CONTAINER_NAME
